foreach(file vfs.c fs_embedded.c strutils.c nuts_bolts.c)
  list(APPEND core_sources "${COMPONENT_CORE_ROOT}/${file}")
  set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()

set(fatfs "${HARNESS_ROOT}/drivers/stm32f4/FatFs")

 "${HARNESS_ROOT}/plugins/networking/http_upload.c" "${HARNESS_ROOT}/plugins/networking/multipartparser.c"
 "${HARNESS_ROOT}/plugins/sdcard/fs_fatfs.c" "${fatfs}/ff.c" "${fatfs}/ffunicode.c"
 "${HARNESS_ROOT}/deps/lwip/src/core/def.c" "${HARNESS_ROOT}/tests/plugins/sdcard_fatfs/fatfs_host.c")
foreach(file vfs.c fs_ram.c stream_json.c strutils.c nuts_bolts.c)
 list(APPEND files_sources "${COMPONENT_CORE_ROOT}/${file}")
 set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/sdcard/fs_fatfs.c" "${fatfs}/ff.c" "${fatfs}/ffunicode.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
set_source_files_properties("${HARNESS_ROOT}/plugins/webui/fs_handlers.c" PROPERTIES COMPILE_OPTIONS "-Wno-error;-Wno-unused-parameter")
set_source_files_properties("${HARNESS_ROOT}/plugins/networking/http_upload.c" PROPERTIES COMPILE_OPTIONS -Wno-unused-parameter)
foreach(mode 2 3)
 foreach(behavior listing directories delete quiet download upload_file upload_fragmented upload_empty upload_maintenance upload_unavailable hidden_files)
  set(target plugin_webui.files_v${mode}_${behavior})
  component_case(${target} files/${behavior}.c files/files_host.c ${files_sources})
  set_tests_properties(${target} PROPERTIES ENVIRONMENT "TZ=UTC0")
  target_include_directories(${target} BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/files")
  target_include_directories(${target} PRIVATE "${HARNESS_ROOT}/tests/plugins/sdcard_fatfs" "${fatfs}" "${HARNESS_ROOT}/plugins/sdcard" "${HARNESS_ROOT}/plugins" "${HARNESS_ROOT}/plugins/webui" "${HARNESS_ROOT}/plugins/networking" "${HARNESS_ROOT}/tests/plugins/networking" "${HARNESS_ROOT}/deps/lwip/src/include")
  target_compile_definitions(${target} PRIVATE WEBUI_ENABLE=${mode} STM32_PLATFORM SDCARD_ENABLE=1 HTTP_ENABLE=1 ETHERNET_ENABLE=1)
  if(MINGW)
   target_compile_options(${target} PRIVATE "SHELL:-include \"${HARNESS_ROOT}/tests/include/windows_string_compat.h\"" "SHELL:-include windows.h")
  endif()
 endforeach()
endforeach()
foreach(mode 0 1 2)
  set(behaviors provider)
  if(mode EQUAL 0)
    list(APPEND behaviors argument_values argument_presence boolean_values bare_boolean trim_middle trim_ends command_arguments)
  elseif(mode EQUAL 1)
    list(APPEND behaviors compressed_asset readonly_assets)
  else()
    list(APPEND behaviors ap_asset)
  endif()
  foreach(behavior IN LISTS behaviors)
    set(target plugin_webui.m${mode}_${behavior})
    component_case(${target} ${behavior}.c webui_host.c "${HARNESS_ROOT}/plugins/webui/args.c" "${HARNESS_ROOT}/plugins/webui/webui_files.c" ${core_sources})
    target_include_directories(${target} PRIVATE "${HARNESS_ROOT}/plugins/webui")
    if(mode EQUAL 0)
      target_compile_definitions(${target} PRIVATE WEBUI_INFLASH=0 WIFI_SOFTAP=0)
    elseif(mode EQUAL 1)
      target_compile_definitions(${target} PRIVATE WEBUI_INFLASH=1 WIFI_SOFTAP=0)
    else()
      target_compile_definitions(${target} PRIVATE WEBUI_INFLASH=1 WIFI_SOFTAP=1)
    endif()
    if(MINGW)
      target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
    endif()
  endforeach()
endforeach()
