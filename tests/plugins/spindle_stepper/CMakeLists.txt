cmake_minimum_required(VERSION 3.20)
project(grblhal_spindle_stepper_tests LANGUAGES C)
include(../../components.cmake)
set(sources)
foreach(file spindle_control.c nuts_bolts.c)
  list(APPEND sources "${COMPONENT_CORE_ROOT}/${file}")
  set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(behavior run_stop reverse enable_mask axis_control rotary settings reset no_driver no_clock)
  set(target plugin_spindle_stepper.${behavior})
  component_case(${target} ${behavior}.c stepper_host.c "${HARNESS_ROOT}/plugins/spindle/stepper.c" "${COMPONENT_CORE_ROOT}/stepper2.c"
    ${sources})
  if(MINGW)
    target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
  endif()
  target_compile_definitions(${target} PRIVATE N_AXIS=4 SPINDLE_ENABLE=524288)
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/spindle/stepper.c" "${COMPONENT_CORE_ROOT}/stepper2.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
