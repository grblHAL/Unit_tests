cmake_minimum_required(VERSION 3.20)
project(grblhal_spindle_pwm_tests LANGUAGES C)
include(../../components.cmake)
set(sources)
foreach(file spindle_control.c ioports.c crossbar.c nuts_bolts.c crc.c strutils.c)
  list(APPEND sources "${COMPONENT_CORE_ROOT}/${file}")
  set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(direction 0 1)
  if(direction)
    set(mask 32768)
  else()
    set(mask 65536)
  endif()
  foreach(behavior no_analog no_pwm no_storage)
    set(target plugin_spindle_pwm.dir${direction}_${behavior})
    component_case(${target} ${behavior}.c pwm_host.c "${HARNESS_ROOT}/plugins/spindle/pwm.c"
      "${HARNESS_ROOT}/plugins/eeprom/eeprom_24AAxxx.c" "${HARNESS_ROOT}/tests/plugins/eeprom/eeprom_host.c" ${sources})
    target_include_directories(${target} PRIVATE "${HARNESS_ROOT}/tests/plugins/eeprom" "${HARNESS_ROOT}/plugins/eeprom")
    if(MINGW)
      target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
    endif()
    target_compile_definitions(${target} PRIVATE N_SPINDLE=2 SPINDLE_ENABLE=${mask} TEST_DIRECTION=${direction} EEPROM_ENABLE=128 EEPROM_IS_FRAM=1 NVS_SIZE=8192 NVS_SIZE_MAX=16384)
  endforeach()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/spindle/pwm.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
