cmake_minimum_required(VERSION 3.20)
project(grblhal_virtual_inputs_tests LANGUAGES C)
include(../../components.cmake)
foreach(topology 0 1)
  set(behaviors probe_toward probe_away virtual_toolsetter physical_toolsetter)
  if(topology EQUAL 0)
    list(APPEND behaviors inversion public_state selected_probe2 realtime_counters command_set command_rejection counter_reset reset_lifecycle unknown_chaining idempotent_init optional_hooks options missing_probe)
  endif()
  foreach(behavior IN LISTS behaviors)
    component_case(plugin_virtual_inputs.t${topology}_${behavior} ${behavior}.c vinput_host.c "${COMPONENT_CORE_ROOT}/probe.c" "${COMPONENT_CORE_ROOT}/nuts_bolts.c" "${HARNESS_ROOT}/plugins/virtual_inputs/virtual_inputs.c")
    target_compile_definitions(plugin_virtual_inputs.t${topology}_${behavior} PRIVATE VIRTUAL_INPUTS_ENABLE=1 VI_SEPARATE=${topology})
    target_include_directories(plugin_virtual_inputs.t${topology}_${behavior} PRIVATE "${HARNESS_ROOT}/plugins/virtual_inputs")
  endforeach()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/virtual_inputs/virtual_inputs.c" "${COMPONENT_CORE_ROOT}/probe.c" PROPERTIES COMPILE_OPTIONS -Wno-unused-parameter)
