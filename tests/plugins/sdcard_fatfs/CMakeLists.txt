cmake_minimum_required(VERSION 3.20)
project(grblhal_sdcard_fatfs_tests LANGUAGES C)
include(../../components.cmake)
set(fatfs "${HARNESS_ROOT}/drivers/stm32f4/FatFs")
set(sources "${HARNESS_ROOT}/plugins/sdcard/fs_fatfs.c" "${fatfs}/ff.c" "${fatfs}/ffunicode.c"
  "${COMPONENT_CORE_ROOT}/vfs.c" "${COMPONENT_CORE_ROOT}/strutils.c" "${COMPONENT_CORE_ROOT}/nuts_bolts.c")
set_source_files_properties(${sources} PROPERTIES COMPILE_OPTIONS -Wno-error)
foreach(behavior roundtrip directory mutations metadata cwd_space read_error missing)
  set(target plugin_sdcard_fatfs.${behavior})
  component_case(${target} ${behavior}.c fatfs_host.c ${sources})
  target_include_directories(${target} PRIVATE "${fatfs}" "${HARNESS_ROOT}/plugins/sdcard")
  target_compile_definitions(${target} PRIVATE STM32_PLATFORM)
  if(MINGW)
    # FatFs' native Windows branch must load platform headers before CORE's On macro.
    target_compile_options(${target} PRIVATE "SHELL:-include \"${HARNESS_ROOT}/tests/include/windows_string_compat.h\"" "SHELL:-include windows.h")
  endif()
endforeach()

foreach(behavior mount detect automount_failure early_mount absent_pin custom_events options)
  set(target plugin_sdcard_fatfs.sd_${behavior})
  component_case(${target} sd_${behavior}.c sd_controller_host.c fatfs_host.c "${HARNESS_ROOT}/plugins/sdcard/sdcard.c" ${sources})
  target_include_directories(${target} PRIVATE "${fatfs}" "${HARNESS_ROOT}/plugins/sdcard")
  target_compile_definitions(${target} PRIVATE STM32_PLATFORM TEST_SDCARD=1)
  if(MINGW)
    target_compile_options(${target} PRIVATE "SHELL:-include \"${HARNESS_ROOT}/tests/include/windows_string_compat.h\"" "SHELL:-include windows.h")
  endif()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/sdcard/sdcard.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
