foreach(chip 3221 4725)
  foreach(behavior startup descriptors unavailable_start unavailable_probe)
    set(target plugins_misc.mcp${chip}_${behavior})
    if(behavior MATCHES "^unavailable_")
      set(test_source analog/unavailable.c)
    else()
      set(test_source analog/${behavior}.c)
    endif()
    component_case(${target} ${test_source} analog/analog_host.c "${HARNESS_ROOT}/plugins/misc/mcp${chip}.c")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/analog")

    endif()
  endforeach()
endforeach()
