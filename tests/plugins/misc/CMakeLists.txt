foreach(chip 3221 4725)
  foreach(behavior startup descriptors unavailable_start unavailable_probe)
    set(target plugins_misc.mcp${chip}_${behavior})
    if(behavior MATCHES "^unavailable_")
      set(test_source analog/unavailable.c)
    else()
      set(test_source analog/${behavior}.c)
    endif()
    component_case(${target} ${test_source} analog/analog_host.c "${HARNESS_ROOT}/plugins/misc/mcp${chip}.c")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/analog")
    target_compile_definitions(${target} PRIVATE MCP${chip}_ENABLE=1)
    if(behavior STREQUAL "unavailable_start")
      target_compile_definitions(${target} PRIVATE FAIL_START=1)
    endif()
  endforeach()
endforeach()
foreach(behavior adc_samples dac_writes dac_send_failure)
  if(behavior STREQUAL "adc_samples")
    set(chip 3221)
  else()
    set(chip 4725)
  endif()
  set(target plugins_misc.${behavior})
  component_case(${target} analog/${behavior}.c analog/analog_host.c "${HARNESS_ROOT}/plugins/misc/mcp${chip}.c")
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/analog")
  target_compile_definitions(${target} PRIVATE MCP${chip}_ENABLE=1)
endforeach()
