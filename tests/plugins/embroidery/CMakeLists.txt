cmake_minimum_required(VERSION 3.20)
project(grblhal_embroidery_readers LANGUAGES C)
include(../../components.cmake)
set(core_sources vfs.c fs_embedded.c strutils.c nuts_bolts.c)
set(sources)
foreach(file IN LISTS core_sources)
 list(APPEND sources "${COMPONENT_CORE_ROOT}/${file}")
 set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(behavior dst_metadata dst_stitches dst_large_moves dst_jump_stop dst_end dst_probe pes_metadata pes_short_moves pes_long_moves pes_jump_trim pes_color_change pes_probe format_fallback)
 component_case(plugin_embroidery_readers.${behavior} ${behavior}.c embroidery_host.c "${HARNESS_ROOT}/plugins/embroidery/tajima.c" "${HARNESS_ROOT}/plugins/embroidery/brother.c" ${sources})
 target_include_directories(plugin_embroidery_readers.${behavior} PRIVATE "${HARNESS_ROOT}/plugins/embroidery")
 if(MINGW)
  target_compile_options(plugin_embroidery_readers.${behavior} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
 endif()
 target_compile_definitions(plugin_embroidery_readers.${behavior} PRIVATE EMBROIDERY_ENABLE=1)
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/embroidery/tajima.c" "${HARNESS_ROOT}/plugins/embroidery/brother.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
