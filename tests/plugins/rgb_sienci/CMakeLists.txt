cmake_minimum_required(VERSION 3.20)
project(grblhal_sienci_rgb_tests LANGUAGES C)
include(../../components.cmake)
foreach(mode 2 3)
  set(behaviors absent_hooks optional_driver_callbacks offboard_green startup states validation manual_overrides brightness missing_parameters job_completed options_chaining initialization_failure stale_tool_color tool_selected)
  if(mode EQUAL 3)
    list(APPEND behaviors animations busy_animation animation_override)
  endif()
  foreach(behavior IN LISTS behaviors)
    set(target plugin_rgb_sienci.m${mode}_${behavior})
    component_case(${target} ${behavior}.c rgb_host.c "${COMPONENT_CORE_ROOT}/nuts_bolts.c" "${HARNESS_ROOT}/plugins/rgb_sienci/rgb.c" "${HARNESS_ROOT}/plugins/rgb_sienci/rgbv2.c")
    target_compile_definitions(${target} PRIVATE STATUS_LIGHT_ENABLE=${mode})
  endforeach()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/rgb_sienci/rgb.c" "${HARNESS_ROOT}/plugins/rgb_sienci/rgbv2.c" PROPERTIES COMPILE_OPTIONS -Wno-unused-parameter)

set_tests_properties(plugin_rgb_sienci.m2_stale_tool_color PROPERTIES WILL_FAIL TRUE LABELS known_bug)
