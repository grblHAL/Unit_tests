cmake_minimum_required(VERSION 3.20)
project(grblhal_laser_overdrive_tests LANGUAGES C)
include(../../components.cmake)
set(sources)
foreach(file spindle_control.c ioports.c crossbar.c nuts_bolts.c crc.c strutils.c)
  list(APPEND sources "${COMPONENT_CORE_ROOT}/${file}")
  set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(behavior commands validation availability settings reset program_end hooks no_selection)
  set(target plugin_laser_overdrive.${behavior})
  component_case(${target} ${behavior}.c overdrive_host.c "${HARNESS_ROOT}/plugins/laser/co2.c" ${sources})
  if(MINGW)
    target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
  endif()
  target_compile_definitions(${target} PRIVATE LASER_OVD_ENABLE=1 N_SPINDLE=2)
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/laser/co2.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
