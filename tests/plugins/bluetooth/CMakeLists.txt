cmake_minimum_required(VERSION 3.20)
project(grblhal_bluetooth_tests LANGUAGES C)
include(../../components.cmake)
set(core_sources ioports.c crossbar.c stream.c stream_passthru.c vfs.c strutils.c nuts_bolts.c crc.c)
set(sources)
foreach(file IN LISTS core_sources)
 list(APPEND sources "${COMPONENT_CORE_ROOT}/${file}")
 set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(behavior config_timeout initially_connected port_settings claimed_pin default_state_pin autoconfig custom_name config_fail1 config_fail2 config_fail3 config_fail4 connected disabled no_inputs no_uart no_storage no_irq_pin reports persist_write persist_read persist_cleanup)
 set(case_source ${behavior}.c)
 if(behavior MATCHES "^config_fail([1-4])$")
  set(case_source config_failure.c)
  set(fail_command ${CMAKE_MATCH_1})
 endif()
 component_case(plugin_bluetooth.${behavior} ${case_source} bt_host.c "${HARNESS_ROOT}/plugins/bluetooth/hc_05.c" "${HARNESS_ROOT}/plugins/eeprom/eeprom_24AAxxx.c" "${HARNESS_ROOT}/tests/plugins/eeprom/eeprom_host.c" ${sources})
 target_include_directories(plugin_bluetooth.${behavior} PRIVATE "${HARNESS_ROOT}/tests/plugins/eeprom" "${HARNESS_ROOT}/plugins/eeprom")
 target_compile_definitions(plugin_bluetooth.${behavior} PRIVATE BLUETOOTH_ENABLE=2 BLUETOOTH_STREAM=1 MPG_BUFFER_SIZE=0 EEPROM_ENABLE=128 EEPROM_IS_FRAM=1 NVS_SIZE=8192 NVS_SIZE_MAX=16384)
 if(behavior MATCHES "^config_fail")
  target_compile_definitions(plugin_bluetooth.${behavior} PRIVATE BT_FAIL_COMMAND=${fail_command})
 endif()
 if(MINGW)
  target_compile_options(plugin_bluetooth.${behavior} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
 endif()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/bluetooth/hc_05.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bt-persistent")
set_tests_properties(plugin_bluetooth.persist_write PROPERTIES FIXTURES_SETUP bt_nvs)
set_tests_properties(plugin_bluetooth.persist_read PROPERTIES FIXTURES_REQUIRED bt_nvs)
set_tests_properties(plugin_bluetooth.persist_cleanup PROPERTIES FIXTURES_CLEANUP bt_nvs)
set_tests_properties(plugin_bluetooth.persist_write plugin_bluetooth.persist_read plugin_bluetooth.persist_cleanup PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bt-persistent")

set_tests_properties(plugin_bluetooth.default_state_pin PROPERTIES WILL_FAIL TRUE LABELS known_bug)
set_tests_properties(plugin_bluetooth.initially_connected PROPERTIES WILL_FAIL TRUE LABELS known_bug)
