cmake_minimum_required(VERSION 3.20)
project(grblhal_networking_tests LANGUAGES C)
include(../../components.cmake)
if(NOT EXISTS "${HARNESS_ROOT}/deps/lwip/src/include/lwip/opt.h")
 message(FATAL_ERROR "Fetch the networking source lock including lwIP first")
endif()
foreach(behavior events enumeration stop_enumeration missing_provider interface_report option_report wifi_report stream_tracking services mac_addresses initialization)
 component_case(plugin_networking.${behavior} ${behavior}.c net_host.c "${HARNESS_ROOT}/plugins/networking/networking.c" "${COMPONENT_CORE_ROOT}/nuts_bolts.c")
 target_include_directories(plugin_networking.${behavior} PRIVATE "${HARNESS_ROOT}/plugins/networking" "${HARNESS_ROOT}/deps/lwip/src/include")
 target_compile_definitions(plugin_networking.${behavior} PRIVATE ETHERNET_ENABLE=1 WIFI_ENABLE=1 TELNET_ENABLE=1 WEBSOCKET_ENABLE=1 DNS_ENABLE=1 MDNS_ENABLE=1 MQTT_ENABLE=0 HTTP_ENABLE=0 FTP_ENABLE=0 SSDP_ENABLE=0 FS_ENABLE=0 MODBUS_ENABLE=0)
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/networking/networking.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
