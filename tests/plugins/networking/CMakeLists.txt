cmake_minimum_required(VERSION 3.20)
project(grblhal_networking_tests LANGUAGES C)
include(../../components.cmake)
if(NOT EXISTS "${HARNESS_ROOT}/deps/lwip/src/include/lwip/opt.h")
 message(FATAL_ERROR "Fetch the networking source lock including lwIP first")
endif()
foreach(behavior events enumeration stop_enumeration missing_provider interface_report option_report wifi_report stream_tracking services mac_addresses initialization)
 component_case(plugin_networking.${behavior} ${behavior}.c net_host.c "${HARNESS_ROOT}/plugins/networking/networking.c" "${COMPONENT_CORE_ROOT}/nuts_bolts.c")
 target_include_directories(plugin_networking.${behavior} PRIVATE "${HARNESS_ROOT}/plugins/networking" "${HARNESS_ROOT}/deps/lwip/src/include")
 target_compile_definitions(plugin_networking.${behavior} PRIVATE ETHERNET_ENABLE=1 WIFI_ENABLE=1 TELNET_ENABLE=1 WEBSOCKET_ENABLE=1 DNS_ENABLE=1 MDNS_ENABLE=1 MQTT_ENABLE=0 HTTP_ENABLE=0 FTP_ENABLE=0 SSDP_ENABLE=0 FS_ENABLE=0 MODBUS_ENABLE=0)
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/networking/networking.c" PROPERTIES COMPILE_OPTIONS -Wno-error)

set(telnet_core_sources)
foreach(file stream.c stream_passthru.c vfs.c strutils.c nuts_bolts.c)
 list(APPEND telnet_core_sources "${COMPONENT_CORE_ROOT}/${file}")
 set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(behavior session input_chain realtime cancel_flush output output_retry mpg_discard remote_close link_down bind_failure output_full)
 set(target plugin_networking.telnet_${behavior})
 component_case(${target} telnet/${behavior}.c telnet/telnet_host.c "${HARNESS_ROOT}/plugins/networking/telnetd.c" ${telnet_core_sources})
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/telnet" "${HARNESS_ROOT}/plugins/networking" "${HARNESS_ROOT}/deps/lwip/src/include")
 target_compile_definitions(${target} PRIVATE ETHERNET_ENABLE=1 TELNET_ENABLE=1 WEBSOCKET_ENABLE=0 MQTT_ENABLE=0 HTTP_ENABLE=0 FTP_ENABLE=0 MDNS_ENABLE=0 MODBUS_ENABLE=0 MPG_BUFFER_SIZE=0 RX_BUFFER_SIZE=32 TX_BUFFER_SIZE=32)
 if(MINGW)
  target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
 endif()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/networking/telnetd.c" PROPERTIES COMPILE_OPTIONS -Wno-unused-parameter)

foreach(behavior file fragmented form_and_file binary_file empty_file optional_callbacks)
 set(target plugin_networking.multipart_${behavior})
 component_case(${target} multipart/${behavior}.c multipart/multipart_host.c "${HARNESS_ROOT}/plugins/networking/multipartparser.c")
 target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/multipart" "${HARNESS_ROOT}/plugins/networking")
endforeach()
