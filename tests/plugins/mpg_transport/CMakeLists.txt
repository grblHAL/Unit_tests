cmake_minimum_required(VERSION 3.20)
project(grblhal_mpg_transport_tests LANGUAGES C)
include(../../components.cmake)
foreach(mode 0 1)
  if(mode EQUAL 0)
    set(behaviors plain_io cancel_and_flush byte_uart realtime_proxy rx_mute byte_output)
  else()
    set(behaviors initialization wire_crc receive_data duplicate_delivery channels fragmented_delivery acknowledged_output timeout_retry nak_retry retry_budget backpressure chunked_output soft_reset baud_change claim_failure registration_failure no_nvs settings_rejected statistics_reset settings_during_output pending_transport discard_output byte_uart settings_restore settings_payload settings_attempts delay_poll realtime_proxy rx_mute byte_output)
  endif()
  foreach(behavior IN LISTS behaviors)
    set(target plugin_mpg_transport.m${mode}_${behavior})
    component_case(${target} ${behavior}.c mpg_host.c "${COMPONENT_CORE_ROOT}/nuts_bolts.c" "${HARNESS_ROOT}/plugins/mpg_transport/mpg_transport.c")
    target_compile_definitions(${target} PRIVATE MPG_ENABLE=3 MPG_STREAM=1 RX_BUFFER_SIZE=256 MPG_TRANSPORT_DEFAULT_TRANSPORT=${mode})
    target_include_directories(${target} PRIVATE "${HARNESS_ROOT}/plugins/mpg_transport")
  endforeach()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/mpg_transport/mpg_transport.c" PROPERTIES COMPILE_OPTIONS "-Wno-unused-parameter")
