cmake_minimum_required(VERSION 3.20)
project(grblhal_spindle_clone_tests LANGUAGES C)
include(../../components.cmake)
set(sources)
foreach(file spindle_control.c ioports.c crossbar.c nuts_bolts.c crc.c strutils.c)
  list(APPEND sources "${COMPONENT_CORE_ROOT}/${file}")
  set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(behavior primary clone switching settings no_base no_direction no_update no_storage)
  set(target plugin_spindle_clone.${behavior})
  set(test_source ${behavior}.c)
  set(reject 0)
  if(behavior MATCHES "^no_")
    set(test_source rejected.c)
    if(behavior STREQUAL "no_base")
      set(reject 1)
    elseif(behavior STREQUAL "no_direction")
      set(reject 2)
    elseif(behavior STREQUAL "no_update")
      set(reject 3)
    else()
      set(reject 4)
    endif()
  endif()
  component_case(${target} ${test_source} clone_host.c "${HARNESS_ROOT}/plugins/spindle/pwm_clone.c"
    "${HARNESS_ROOT}/plugins/eeprom/eeprom_24AAxxx.c" "${HARNESS_ROOT}/tests/plugins/eeprom/eeprom_host.c" ${sources})
  target_include_directories(${target} PRIVATE "${HARNESS_ROOT}/tests/plugins/eeprom" "${HARNESS_ROOT}/plugins/eeprom")
  if(MINGW)
    target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
  endif()
  target_compile_definitions(${target} PRIVATE N_SPINDLE=2 DEFAULT_SPINDLE1_RPM_MAX=2000 SPINDLE_ENABLE=133120 TEST_REJECT=${reject} EEPROM_ENABLE=128 EEPROM_IS_FRAM=1 NVS_SIZE=8192 NVS_SIZE_MAX=16384)
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/spindle/pwm_clone.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
