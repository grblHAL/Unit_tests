cmake_minimum_required(VERSION 3.20)
project(grblhal_keypad_tests LANGUAGES C)
include(../../components.cmake)
foreach(mode 1 2)
 foreach(behavior jog_modes diagonal callbacks preview legacy_control extended_control overrides extended_overrides no_storage no_transport release_before_foreground persist_write persist_read)
  component_case(plugin_keypad.${mode}_${behavior} ${behavior}.c key_host.c "${HARNESS_ROOT}/plugins/keypad/keypad.c" "${COMPONENT_CORE_ROOT}/override.c" "${COMPONENT_CORE_ROOT}/crc.c" "${COMPONENT_CORE_ROOT}/nuts_bolts.c")
  target_compile_options(plugin_keypad.${mode}_${behavior} PRIVATE -fsigned-char)
  target_include_directories(plugin_keypad.${mode}_${behavior} PRIVATE "${HARNESS_ROOT}/plugins/keypad")
  target_compile_definitions(plugin_keypad.${mode}_${behavior} PRIVATE KEYPAD_ENABLE=${mode} KEYPAD_STREAM=1 MPG_ENABLE=0)
 endforeach()
 file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${mode}-persistent")
 set_tests_properties(plugin_keypad.${mode}_persist_write PROPERTIES FIXTURES_SETUP key_${mode}_nvs)
 set_tests_properties(plugin_keypad.${mode}_persist_read PROPERTIES FIXTURES_REQUIRED key_${mode}_nvs)
 set_tests_properties(plugin_keypad.${mode}_persist_write plugin_keypad.${mode}_persist_read PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${mode}-persistent")
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/keypad/keypad.c" PROPERTIES COMPILE_OPTIONS -Wno-error)

foreach(mode 1 2)
 set_tests_properties(plugin_keypad.${mode}_extended_control plugin_keypad.${mode}_extended_overrides PROPERTIES WILL_FAIL TRUE LABELS known_bug)
 foreach(behavior extended_control extended_overrides)
  component_case(plugin_keypad.${mode}_unsigned_${behavior} ${behavior}.c key_host.c "${HARNESS_ROOT}/plugins/keypad/keypad.c" "${COMPONENT_CORE_ROOT}/override.c" "${COMPONENT_CORE_ROOT}/crc.c" "${COMPONENT_CORE_ROOT}/nuts_bolts.c")
  target_include_directories(plugin_keypad.${mode}_unsigned_${behavior} PRIVATE "${HARNESS_ROOT}/plugins/keypad")
  target_compile_definitions(plugin_keypad.${mode}_unsigned_${behavior} PRIVATE KEYPAD_ENABLE=${mode} KEYPAD_STREAM=1 MPG_ENABLE=0)
  target_compile_options(plugin_keypad.${mode}_unsigned_${behavior} PRIVATE -funsigned-char)
 endforeach()
endforeach()

foreach(behavior states outputs no_bus no_probe reports optional_hooks write_recovery)
 set(target plugin_keypad.led_${behavior})
 component_case(${target} led_${behavior}.c led_host.c "${HARNESS_ROOT}/plugins/keypad/display/i2c_leds.c")
 target_compile_definitions(${target} PRIVATE DISPLAY_ENABLE=17)
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/keypad/display/i2c_leds.c" PROPERTIES COMPILE_OPTIONS -Wno-error)

set(macro_sources "${HARNESS_ROOT}/plugins/keypad/macros.c"
 "${HARNESS_ROOT}/plugins/eeprom/eeprom_24AAxxx.c" "${HARNESS_ROOT}/tests/plugins/eeprom/eeprom_host.c"
 "${COMPONENT_CORE_ROOT}/crc.c" "${COMPONENT_CORE_ROOT}/strutils.c" "${COMPONENT_CORE_ROOT}/nuts_bolts.c")
set_source_files_properties("${HARNESS_ROOT}/plugins/keypad/macros.c" "${COMPONENT_CORE_ROOT}/strutils.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
foreach(behavior blocks separator keypress state_guard reset error no_storage save_load empty keypress_signed)
 set(target plugin_keypad.macro_${behavior})
 if(behavior STREQUAL "keypress_signed")
  set(case_source macro_keypress.c)
  set(char_flag -fsigned-char)
 else()
  set(case_source macro_${behavior}.c)
  set(char_flag -funsigned-char)
 endif()
 component_case(${target} ${case_source} macro_host.c ${macro_sources})
 target_include_directories(${target} PRIVATE "${HARNESS_ROOT}/plugins/keypad" "${HARNESS_ROOT}/tests/plugins/eeprom" "${HARNESS_ROOT}/plugins/eeprom")
 target_compile_definitions(${target} PRIVATE KEYPAD_ENABLE=2 MACROS_ENABLE=2 EEPROM_ENABLE=128 EEPROM_IS_FRAM=1 NVS_SIZE=8192 NVS_SIZE_MAX=16384)
 target_compile_options(${target} PRIVATE ${char_flag} "SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/macro_platform.h\"")
endforeach()

set_tests_properties(plugin_keypad.macro_keypress_signed PROPERTIES WILL_FAIL TRUE LABELS known_bug)

foreach(file ioports.c crossbar.c)
 set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(behavior execution state_guard actions irq_failure no_ports)
 set(target plugin_keypad.macro_aux_${behavior})
 component_case(${target} macro_aux_${behavior}.c macro_host.c ${macro_sources}
 "${COMPONENT_CORE_ROOT}/ioports.c" "${COMPONENT_CORE_ROOT}/crossbar.c")
 target_include_directories(${target} PRIVATE "${HARNESS_ROOT}/plugins/keypad" "${HARNESS_ROOT}/tests/plugins/eeprom" "${HARNESS_ROOT}/plugins/eeprom")
 if(behavior STREQUAL "no_ports")
  set(macro_mode 1)
 else()
  set(macro_mode 3)
 endif()
 target_compile_definitions(${target} PRIVATE KEYPAD_ENABLE=2 MACROS_ENABLE=${macro_mode} MACRO_1_AUXIN=0 MACRO_2_AUXIN=1 EEPROM_ENABLE=128 EEPROM_IS_FRAM=1 NVS_SIZE=8192 NVS_SIZE_MAX=16384)
 target_compile_options(${target} PRIVATE -funsigned-char "SHELL:-include \"${CMAKE_CURRENT_SOURCE_DIR}/macro_platform.h\"")
endforeach()
