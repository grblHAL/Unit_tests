cmake_minimum_required(VERSION 3.20)
project(grblhal_keypad_tests LANGUAGES C)
include(../../components.cmake)
foreach(mode 1 2)
 foreach(behavior jog_modes diagonal callbacks preview legacy_control extended_control overrides extended_overrides no_storage no_transport release_before_foreground persist_write persist_read)
  component_case(plugin_keypad.${mode}_${behavior} ${behavior}.c key_host.c "${HARNESS_ROOT}/plugins/keypad/keypad.c" "${COMPONENT_CORE_ROOT}/override.c" "${COMPONENT_CORE_ROOT}/crc.c" "${COMPONENT_CORE_ROOT}/nuts_bolts.c")
  target_compile_options(plugin_keypad.${mode}_${behavior} PRIVATE -fsigned-char)
  target_include_directories(plugin_keypad.${mode}_${behavior} PRIVATE "${HARNESS_ROOT}/plugins/keypad")
  target_compile_definitions(plugin_keypad.${mode}_${behavior} PRIVATE KEYPAD_ENABLE=${mode} KEYPAD_STREAM=1 MPG_ENABLE=0)
 endforeach()
 file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${mode}-persistent")
 set_tests_properties(plugin_keypad.${mode}_persist_write PROPERTIES FIXTURES_SETUP key_${mode}_nvs)
 set_tests_properties(plugin_keypad.${mode}_persist_read PROPERTIES FIXTURES_REQUIRED key_${mode}_nvs)
 set_tests_properties(plugin_keypad.${mode}_persist_write plugin_keypad.${mode}_persist_read PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${mode}-persistent")
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/keypad/keypad.c" PROPERTIES COMPILE_OPTIONS -Wno-error)

foreach(mode 1 2)
 set_tests_properties(plugin_keypad.${mode}_extended_control plugin_keypad.${mode}_extended_overrides PROPERTIES WILL_FAIL TRUE LABELS known_bug)
 foreach(behavior extended_control extended_overrides)
  component_case(plugin_keypad.${mode}_unsigned_${behavior} ${behavior}.c key_host.c "${HARNESS_ROOT}/plugins/keypad/keypad.c" "${COMPONENT_CORE_ROOT}/override.c" "${COMPONENT_CORE_ROOT}/crc.c" "${COMPONENT_CORE_ROOT}/nuts_bolts.c")
  target_include_directories(plugin_keypad.${mode}_unsigned_${behavior} PRIVATE "${HARNESS_ROOT}/plugins/keypad")
  target_compile_definitions(plugin_keypad.${mode}_unsigned_${behavior} PRIVATE KEYPAD_ENABLE=${mode} KEYPAD_STREAM=1 MPG_ENABLE=0)
  target_compile_options(plugin_keypad.${mode}_unsigned_${behavior} PRIVATE -funsigned-char)
 endforeach()
endforeach()
