cmake_minimum_required(VERSION 3.20)
project(grblhal_interlock_tests LANGUAGES C)
include(../../components.cmake)
foreach(behavior settings_readback inverse_disable direction_polarity settings_restore disabled_defaults forward_high forward_low_output forward_multiple stop_allowed unreadable_monitor inverse_running inverse_release port_settings nvs_reload zero_ports init_failure hook_chaining)
  component_case(plugin_interlock.${behavior} ${behavior}.c interlock_host.c "${HARNESS_ROOT}/plugins/interlock/spindle_interlock.c")
  target_compile_definitions(plugin_interlock.${behavior} PRIVATE SPINDLE_INTERLOCK_ENABLE=1)
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/interlock/spindle_interlock.c" PROPERTIES COMPILE_OPTIONS "-Wno-unused-parameter")
if(CMAKE_C_COMPILER_ID STREQUAL "Clang")
  set_property(SOURCE "${HARNESS_ROOT}/plugins/interlock/spindle_interlock.c" APPEND PROPERTY COMPILE_OPTIONS -Wno-error=default-const-init-field-unsafe)
endif()

foreach(fwd 1 2)
  if(fwd EQUAL 1)
    set(inv 2)
  else()
    set(inv 1)
  endif()
  set(target plugin_interlock.slots${fwd}_${inv})
  component_case(${target} slot_layout.c interlock_host.c "${HARNESS_ROOT}/plugins/interlock/spindle_interlock.c")
  target_compile_definitions(${target} PRIVATE SPINDLE_INTERLOCK_ENABLE=1 SPINDLE_INTERLOCK_FWD_SLOTS=${fwd} SPINDLE_INTERLOCK_INV_SLOTS=${inv})
endforeach()
