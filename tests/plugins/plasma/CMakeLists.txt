cmake_minimum_required(VERSION 3.20)
project(grblhal_plasma_tests LANGUAGES C)
include(../../components.cmake)
set(core_sources spindle_control.c ioports.c crossbar.c stepper2.c pid.c vfs.c fs_embedded.c strutils.c nuts_bolts.c crc.c)
set(sources)
foreach(file IN LISTS core_sources)
 list(APPEND sources "${COMPONENT_CORE_ROOT}/${file}")
 set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
set(boot_cases boot1_write boot1_cleanup boot2_write boot2_cleanup boot3_write boot3_cleanup arc_start_stop arc_failure arc_retry arc_lost updown_motion updown_vad active_reset)
foreach(behavior ${boot_cases} materials validation comments no_output no_inputs no_storage linuxcnc_import sheetcam_import overwrite comment_rejection enumeration_stop reports voltage_corrections voltage_vad voltage_reset voltage_report voltage_arc_lost)
 set(case_source ${behavior}.c)
 if(behavior MATCHES "^boot[123]_write$")
  set(case_source boot_write.c)
 elseif(behavior MATCHES "^boot[123]_cleanup$")
  set(case_source boot_cleanup.c)
 endif()
 component_case(plugin_plasma.${behavior} ${case_source} plasma_host.c plasma_runtime.c "${HARNESS_ROOT}/plugins/plasma/thc.c" "${HARNESS_ROOT}/plugins/plasma/sheetcam.c" "${HARNESS_ROOT}/plugins/plasma/linuxcnc.c" "${HARNESS_ROOT}/plugins/eeprom/eeprom_24AAxxx.c" "${HARNESS_ROOT}/tests/plugins/eeprom/eeprom_host.c" ${sources})
 target_include_directories(plugin_plasma.${behavior} PRIVATE "${HARNESS_ROOT}/plugins/plasma" "${HARNESS_ROOT}/tests/plugins/eeprom" "${HARNESS_ROOT}/plugins/eeprom")
 target_compile_definitions(plugin_plasma.${behavior} PRIVATE PLASMA_ENABLE=1 NGC_EXPRESSIONS_ENABLE=0 EEPROM_ENABLE=128 EEPROM_IS_FRAM=1 NVS_SIZE=8192 NVS_SIZE_MAX=16384)
 if(MINGW)
  target_compile_options(plugin_plasma.${behavior} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
 endif()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/plasma/thc.c" "${HARNESS_ROOT}/plugins/plasma/sheetcam.c" "${HARNESS_ROOT}/plugins/plasma/linuxcnc.c" PROPERTIES COMPILE_OPTIONS -Wno-error)

foreach(mode 1 2 3)
 file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plasma-${mode}")
 set_tests_properties(plugin_plasma.boot${mode}_write PROPERTIES FIXTURES_SETUP plasma${mode}_nvs)
 set_tests_properties(plugin_plasma.boot${mode}_cleanup PROPERTIES FIXTURES_CLEANUP plasma${mode}_nvs)
 set_tests_properties(plugin_plasma.boot${mode}_write plugin_plasma.boot${mode}_cleanup PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plasma-${mode}")
 target_compile_definitions(plugin_plasma.boot${mode}_write PRIVATE PLASMA_BOOT_MODE=${mode})
endforeach()
foreach(behavior arc_start_stop arc_failure arc_retry arc_lost)
 set_tests_properties(plugin_plasma.${behavior} PROPERTIES FIXTURES_REQUIRED plasma3_nvs WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plasma-3")
endforeach()
foreach(behavior updown_motion updown_vad active_reset)
 set_tests_properties(plugin_plasma.${behavior} PROPERTIES FIXTURES_REQUIRED plasma2_nvs WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plasma-2")
endforeach()

target_compile_definitions(plugin_plasma.boot1_write PRIVATE PLASMA_ANALOG=1)
foreach(behavior voltage_corrections voltage_vad voltage_reset voltage_report voltage_arc_lost)
 target_compile_definitions(plugin_plasma.${behavior} PRIVATE PLASMA_ANALOG=1)
 set_tests_properties(plugin_plasma.${behavior} PROPERTIES FIXTURES_REQUIRED plasma1_nvs WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/plasma-1")
endforeach()
