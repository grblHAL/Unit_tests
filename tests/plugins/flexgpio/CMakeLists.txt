cmake_minimum_required(VERSION 3.20)
project(grblhal_flexgpio_tests LANGUAGES C)
include(../../components.cmake)
foreach(axes 3 6)
  set(behaviors initialization)
  if(axes EQUAL 3)
    list(APPEND behaviors wait_edges enumeration output_packets output_inversion input_reads edge_callbacks change_callbacks io_failures wait_timeout descriptions probe_irq_config pull_mode_rejection no_device no_bus irq_claim_failure)
  endif()
  foreach(behavior IN LISTS behaviors)
    set(target plugin_flexgpio.a${axes}_${behavior})
    component_case(${target} ${behavior}.c flex_host.c "${HARNESS_ROOT}/plugins/flexgpio/flexgpio.c")
    target_compile_definitions(${target} PRIVATE FLEXGPIO_ENABLE=1 FLEXGPIO_IRQ_PIN=1 N_AXIS=${axes})
  endforeach()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/plugins/flexgpio/flexgpio.c" PROPERTIES COMPILE_OPTIONS "-Wno-unused-parameter;-Wno-error=parentheses")
if(CMAKE_C_COMPILER_ID STREQUAL "Clang")
  set_property(SOURCE "${HARNESS_ROOT}/plugins/flexgpio/flexgpio.c" APPEND PROPERTY COMPILE_OPTIONS -Wno-error=default-const-init-field-unsafe)
endif()

foreach(behavior edge_callbacks pull_mode_rejection)
  set_tests_properties(plugin_flexgpio.a3_${behavior} PROPERTIES WILL_FAIL TRUE LABELS known_bug)
endforeach()

if(ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID STREQUAL "Clang")
  foreach(behavior output_inversion input_reads probe_irq_config pull_mode_rejection)
    set_property(TEST plugin_flexgpio.a3_${behavior} APPEND PROPERTY LABELS known_sanitizer_bug)
  endforeach()
endif()
