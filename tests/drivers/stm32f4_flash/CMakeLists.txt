cmake_minimum_required(VERSION 3.20)
project(grblhal_stm32f4_flash_tests LANGUAGES C)
include(../../components.cmake)
# Original linker-address arithmetic is32-bit; retain it rather than truncating host pointers.
foreach(behavior erased_read roundtrip unchanged unlock_failure erase_retry erase_failure program_failure retry_after_failure configured_size)
 component_case(driver_stm32f4_flash.${behavior} "${CMAKE_CURRENT_SOURCE_DIR}/../stm32_flash/${behavior}.c" "${CMAKE_CURRENT_SOURCE_DIR}/../stm32_flash/stm_host.c" "${HARNESS_ROOT}/drivers/stm32f4/Src/flash.c")
 target_include_directories(driver_stm32f4_flash.${behavior} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../stm32_flash")
 target_compile_definitions(driver_stm32f4_flash.${behavior} PRIVATE STM32_FAMILY=4)
 target_compile_options(driver_stm32f4_flash.${behavior} PRIVATE -m32)
 target_link_options(driver_stm32f4_flash.${behavior} PRIVATE -m32)
 if(NOT WIN32)
  target_compile_options(driver_stm32f4_flash.${behavior} PRIVATE -fno-pie)
  target_link_options(driver_stm32f4_flash.${behavior} PRIVATE -no-pie)
 endif()
endforeach()
