cmake_minimum_required(VERSION 3.20)
project(grblhal_sam_serial_tests LANGUAGES C)
include(../../components.cmake)
set(sdk "${HARNESS_ROOT}/deps/arduino-sam")
if(NOT EXISTS "${sdk}/cores/arduino/Arduino.h")
 message(FATAL_ERROR "Fetch the locked Arduino SAM SDK headers first")
endif()
set(core_sources stream.c stream_passthru.c vfs.c strutils.c nuts_bolts.c)
set(sources)
foreach(file IN LISTS core_sources)
 list(APPEND sources "${COMPONENT_CORE_ROOT}/${file}")
 set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(port -1 1)
 foreach(behavior initialization rx_fifo rx_wrap rx_cancel realtime_irq direct_tx queued_tx full_tx_abort blocking_tx configuration ownership disable_primary disable_secondary rx_enable secondary_flush secondary_realtime suspend pinmux)
  if(behavior STREQUAL pinmux AND port LESS 0)
   continue()
  endif()
  set(case_source ${behavior}.c)
  if(behavior MATCHES "^disable_")
   set(case_source disable_rx.c)
  endif()
  set(target driver_sam3x8e_serial.p${port}_${behavior})
  component_case(${target} ${case_source} sam_host.c "${HARNESS_ROOT}/drivers/sam3x8e/grblHAL_Due/src/serial.c" ${sources})
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/sdk" "${HARNESS_ROOT}/drivers/sam3x8e/grblHAL_Due/src")
  target_include_directories(${target} SYSTEM PRIVATE "${sdk}/cores/arduino" "${sdk}/variants/arduino_due_x" "${sdk}/system/libsam" "${sdk}/system/CMSIS/Device/ATMEL" "${sdk}/system/CMSIS/Device/ATMEL/sam3xa/include")
  target_compile_definitions(${target} PRIVATE __SAM3X8E__ OVERRIDE_MY_MACHINE=1 SERIAL_PORT=${port} SERIAL1_PORT=2 MPG_BUFFER_SIZE=0 RX_BUFFER_SIZE=32 TX_BUFFER_SIZE=32)
  if(behavior STREQUAL disable_primary)
   target_compile_definitions(${target} PRIVATE SAM_INSTANCE=0)
  elseif(behavior STREQUAL disable_secondary)
   target_compile_definitions(${target} PRIVATE SAM_INSTANCE=1)
  endif()
  if(MINGW)
   target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
  endif()
  if(NOT WIN32)
   target_compile_definitions(${target} PRIVATE _GNU_SOURCE)
  endif()
 endforeach()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/drivers/sam3x8e/grblHAL_Due/src/serial.c" PROPERTIES COMPILE_OPTIONS -Wno-error)

if(CMAKE_C_COMPILER_ID STREQUAL "Clang")
 set_source_files_properties("${HARNESS_ROOT}/drivers/sam3x8e/grblHAL_Due/src/serial.c" PROPERTIES COMPILE_OPTIONS "-Wno-error;-Wno-error=incompatible-function-pointer-types")
endif()

foreach(port -1 1)
 foreach(behavior disable_primary disable_secondary secondary_realtime)
  set_tests_properties(driver_sam3x8e_serial.p${port}_${behavior} PROPERTIES WILL_FAIL TRUE LABELS known_bug)
 endforeach()
 if(ENABLE_SANITIZERS AND CMAKE_C_COMPILER_ID STREQUAL "Clang")
  set_property(TEST driver_sam3x8e_serial.p${port}_secondary_realtime APPEND PROPERTY LABELS known_sanitizer_bug)
 endif()
endforeach()
set_tests_properties(driver_sam3x8e_serial.p1_pinmux PROPERTIES WILL_FAIL TRUE LABELS known_bug)
