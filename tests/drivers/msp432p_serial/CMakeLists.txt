cmake_minimum_required(VERSION 3.20)
project(grblhal_msp432p_serial_tests LANGUAGES C)
include(../../components.cmake)
set(core_sources)
foreach(file stream.c stream_passthru.c vfs.c strutils.c nuts_bolts.c)
  list(APPEND core_sources "${COMPONENT_CORE_ROOT}/${file}")
  set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
set(original "${HARNESS_ROOT}/drivers/msp432p401r/serial.c")
foreach(behavior mute_preserves_tx initialization rx_order rx_wrap rx_capacity cancel_flush direct_tx queued_tx full_tx_abort blocking_tx disable_rx ownership status_timestamp suspend handler_replace)
  set(target driver_msp432p_serial.${behavior})
  component_case(${target} ${behavior}.c msp_host.c core_boundaries.c "${original}" ${core_sources})
  target_include_directories(${target} PRIVATE "${HARNESS_ROOT}/drivers/msp432p401r")
  target_include_directories(${target} SYSTEM PRIVATE "${HARNESS_ROOT}/deps/msp432-sdk/system/inc" "${HARNESS_ROOT}/deps/msp432-sdk/system/inc/CMSIS")
  target_compile_definitions(${target} PRIVATE __MSP432P401R__ OVERRIDE_MY_MACHINE=1 BOARD_CNC_BOOSTERPACK=1 RX_BUFFER_SIZE=32 TX_BUFFER_SIZE=16 MPG_BUFFER_SIZE=0)
  if(MINGW)
    target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
  endif()
  if(NOT WIN32)
    target_compile_definitions(${target} PRIVATE _GNU_SOURCE)
  endif()
endforeach()
set_source_files_properties("${original}" PROPERTIES COMPILE_OPTIONS "-include;${CMAKE_CURRENT_SOURCE_DIR}/cpu_host.h;-Wno-unused-parameter")

set_tests_properties(driver_msp432p_serial.mute_preserves_tx PROPERTIES WILL_FAIL TRUE LABELS known_bug)
