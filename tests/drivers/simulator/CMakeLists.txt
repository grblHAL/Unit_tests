cmake_minimum_required(VERSION 3.20)
project(grblhal_simulator_tests LANGUAGES C)
include(../../components.cmake)
foreach(behavior erased_storage existing_storage close_without_open)
  component_case(driver_simulator.${behavior} ${behavior}.c "${HARNESS_ROOT}/drivers/simulator/src/eeprom.c")
  target_include_directories(driver_simulator.${behavior} PRIVATE "${HARNESS_ROOT}/drivers/simulator/src")
  if(WIN32)
    target_compile_definitions(driver_simulator.${behavior} PRIVATE PLATFORM_WINDOWS)
  else()
    target_compile_definitions(driver_simulator.${behavior} PRIVATE PLATFORM_LINUX)
  endif()
endforeach()

foreach(behavior block_roundtrip checksum_write checksum_read)
  component_case(driver_simulator.${behavior} ${behavior}.c "${HARNESS_ROOT}/drivers/simulator/src/eeprom.c" "${HARNESS_ROOT}/drivers/simulator/src/grbl_eeprom_extensions.c" "${HARNESS_ROOT}/core/crc.c")
  target_include_directories(driver_simulator.${behavior} PRIVATE "${HARNESS_ROOT}/drivers/simulator/src")
  if(WIN32)
    target_compile_definitions(driver_simulator.${behavior} PRIVATE PLATFORM_WINDOWS)
  else()
    target_compile_definitions(driver_simulator.${behavior} PRIVATE PLATFORM_LINUX)
  endif()
endforeach()
# Upstream retains an unused original-buffer pointer in its block reader.
set_source_files_properties("${HARNESS_ROOT}/drivers/simulator/src/grbl_eeprom_extensions.c" PROPERTIES COMPILE_OPTIONS -Wno-unused-variable)

set_tests_properties(driver_simulator.checksum_write driver_simulator.checksum_read PROPERTIES WILL_FAIL TRUE LABELS known_bug)
