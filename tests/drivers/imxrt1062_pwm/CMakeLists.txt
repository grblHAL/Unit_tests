cmake_minimum_required(VERSION 3.20)
project(grblhal_imxrt_pwm_tests LANGUAGES C)
include(../../components.cmake)
if(NOT EXISTS "${HARNESS_ROOT}/deps/teensy/teensy4/imxrt.h")
 message(FATAL_ERROR "Fetch the locked Teensy SDK headers first")
endif()
foreach(model 40 41 micromod)
 foreach(behavior reconfigure independent_outputs quad_channels availability flex_x flex_ab clocks quad inversion duty_limits ownership board_route)
  set(target driver_imxrt1062_pwm.t${model}_${behavior})
  component_case(${target} ${behavior}.c pwm_host.c "${HARNESS_ROOT}/drivers/imxrt1062/grblHAL_Teensy4/src/pwm.c")
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/sdk" "${HARNESS_ROOT}/drivers/imxrt1062/grblHAL_Teensy4/src")
  target_include_directories(${target} SYSTEM PRIVATE "${HARNESS_ROOT}/deps/teensy/teensy4")
  target_compile_definitions(${target} PRIVATE OVERRIDE_MY_MACHINE=1)
  if(model STREQUAL micromod)
   target_compile_definitions(${target} PRIVATE ARDUINO_TEENSY_MICROMOD)
  else()
   target_compile_definitions(${target} PRIVATE ARDUINO_TEENSY${model})
  endif()
  if(NOT WIN32)
   target_compile_definitions(${target} PRIVATE _GNU_SOURCE)
  endif()
 endforeach()
endforeach()
set_source_files_properties("${HARNESS_ROOT}/drivers/imxrt1062/grblHAL_Teensy4/src/pwm.c" PROPERTIES COMPILE_OPTIONS -Wno-error)
