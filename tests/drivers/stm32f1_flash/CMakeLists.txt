cmake_minimum_required(VERSION 3.20)
project(grblhal_stm32f1_flash_tests LANGUAGES C)
include(../../components.cmake)
foreach(behavior erased_read roundtrip unchanged unlock_failure erase_retry erase_failure program_failure retry_after_failure configured_size)
 set(case_source "${CMAKE_CURRENT_SOURCE_DIR}/../stm32_flash/${behavior}.c")
 if(behavior STREQUAL unlock_failure)
  set(case_source unlock_failure.c)
 endif()
 component_case(driver_stm32f1_flash.${behavior} ${case_source} "${CMAKE_CURRENT_SOURCE_DIR}/../stm32_flash/stm_host.c" "${HARNESS_ROOT}/drivers/stm32f1/Src/flash.c")
 target_include_directories(driver_stm32f1_flash.${behavior} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../stm32_flash")
 target_compile_definitions(driver_stm32f1_flash.${behavior} PRIVATE STM32_FAMILY=1)
 target_compile_options(driver_stm32f1_flash.${behavior} PRIVATE -m32)
 target_link_options(driver_stm32f1_flash.${behavior} PRIVATE -m32)
 if(NOT WIN32)
  target_compile_options(driver_stm32f1_flash.${behavior} PRIVATE -fno-pie)
  target_link_options(driver_stm32f1_flash.${behavior} PRIVATE -no-pie)
 endif()
endforeach()
