cmake_minimum_required(VERSION 3.20)
project(grblhal_stm32h7_flash_tests LANGUAGES C)
include(../../components.cmake)
foreach(model h743 h723)
 foreach(behavior erased_read roundtrip partial_tail unchanged unlock_failure erase_failure program_first_failure program_second_failure retry)
  set(case_source ${behavior}.c)
  if(behavior MATCHES "^program_.*failure$")
   set(case_source program_failure.c)
  endif()
  set(target driver_stm32h7_flash.${model}_${behavior})
  component_case(${target} ${case_source} h7_host.c "${HARNESS_ROOT}/drivers/stm32h7/Src/flash.c")
  target_include_directories(${target} PRIVATE "${HARNESS_ROOT}/drivers/stm32h7/Inc")
  if(model STREQUAL h743)
   target_compile_definitions(${target} PRIVATE STM32H743xx)
  else()
   target_compile_definitions(${target} PRIVATE STM32H723xx)
  endif()
  if(behavior STREQUAL program_first_failure)
   target_compile_definitions(${target} PRIVATE H7_FAIL_WORD=1)
  elseif(behavior STREQUAL program_second_failure)
   target_compile_definitions(${target} PRIVATE H7_FAIL_WORD=2)
  endif()
  target_compile_options(${target} PRIVATE -m32)
  target_link_options(${target} PRIVATE -m32)
  if(NOT WIN32)
   target_compile_options(${target} PRIVATE -fno-pie)
   target_link_options(${target} PRIVATE -no-pie)
  endif()
 endforeach()
endforeach()
