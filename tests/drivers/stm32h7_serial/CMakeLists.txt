cmake_minimum_required(VERSION 3.20)
project(grblhal_stm32h7_serial_tests LANGUAGES C)
include(../../components.cmake)
set(driver "${HARNESS_ROOT}/drivers/stm32h7")
# Reuse SDK data verbatim; the full UART HAL unit requires ARM exclusive-access intrinsics.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${driver}/Drivers/STM32H7xx_HAL_Driver/Src/stm32h7xx_hal_uart.c")
file(STRINGS "${driver}/Drivers/STM32H7xx_HAL_Driver/Src/stm32h7xx_hal_uart.c" prescaler_declaration
 REGEX "^const uint16_t UARTPrescTable\\[12\\] = ")
list(LENGTH prescaler_declaration declaration_count)
if(NOT declaration_count EQUAL 1)
 message(FATAL_ERROR "Expected one original SDK UARTPrescTable declaration")
endif()
list(GET prescaler_declaration 0 prescaler_declaration)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/uart_prescaler.c" CONTENT "#include <stdint.h>\n${prescaler_declaration}\n" @ONLY)
set(core_sources)
foreach(file stream.c stream_passthru.c vfs.c strutils.c nuts_bolts.c)
 list(APPEND core_sources "${COMPONENT_CORE_ROOT}/${file}")
 set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(behavior initialization rx_fifo rx_wrap cancel realtime suspend output output_full configuration ownership rx_disable multi_binary multi_cancel multi_duplex multi_format multi_initialization multi_output_full multi_ownership multi_realtime multi_rx_disable multi_rx_wrap multi_suspend multi_tool_ack h723_initialization)
 set(mcu STM32H743xx)
 set(case_source ${behavior}.c)
 if(behavior STREQUAL h723_initialization)
  set(mcu STM32H723xx)
  set(case_source initialization.c)
 endif()
 set(target driver_stm32h7_serial.${behavior})
 component_case(${target} ${case_source} stmh7_serial_host.c "${HARNESS_ROOT}/tests/drivers/serial_core_host.c" "${driver}/Src/serial.c" "${CMAKE_CURRENT_BINARY_DIR}/uart_prescaler.c" ${core_sources})
 target_include_directories(${target} SYSTEM PRIVATE "${driver}/Inc" "${driver}" "${driver}/Drivers/STM32H7xx_HAL_Driver/Inc" "${driver}/Drivers/CMSIS/Include" "${driver}/Drivers/CMSIS/Device/ST/STM32H7xx/Include")
 target_compile_definitions(${target} PRIVATE ${mcu} USE_HAL_DRIVER OVERRIDE_MY_MACHINE=1 USB_SERIAL_CDC=0 MPG_BUFFER_SIZE=0 RX_BUFFER_SIZE=32 TX_BUFFER_SIZE=32)
 if(behavior MATCHES "^multi_")
  target_compile_definitions(${target} PRIVATE STM_TRIPLEX=1 SERIAL2_PORT=1)
 endif()
 if(MINGW)
  target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
 endif()
endforeach()
set_source_files_properties("${driver}/Src/serial.c" PROPERTIES COMPILE_OPTIONS -Wno-unused-parameter)
