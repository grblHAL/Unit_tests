cmake_minimum_required(VERSION 3.20)
project(grblhal_stm32f1_serial_tests LANGUAGES C)
include(../../components.cmake)
set(driver "${HARNESS_ROOT}/drivers/stm32f1")
set(core_sources)
foreach(file stream.c stream_passthru.c vfs.c strutils.c nuts_bolts.c)
 list(APPEND core_sources "${COMPONENT_CORE_ROOT}/${file}")
 set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(behavior initialization rx_fifo rx_wrap cancel realtime suspend output_full capacity output rx_disable ownership baud)
 set(target driver_stm32f1_serial.${behavior})
 component_case(${target} ${behavior}.c stm1_serial_host.c "${HARNESS_ROOT}/tests/drivers/serial_core_host.c" "${driver}/Src/serial.c" ${core_sources})
 target_include_directories(${target} SYSTEM PRIVATE "${driver}/Inc" "${driver}" "${driver}/Drivers/STM32F1xx_HAL_Driver/Inc" "${driver}/Drivers/CMSIS/Include" "${driver}/Drivers/CMSIS/Device/ST/STM32F1xx/Include")
 target_compile_definitions(${target} PRIVATE STM32F103xB USE_HAL_DRIVER OVERRIDE_MY_MACHINE=1 USB_SERIAL_CDC=0 MPG_BUFFER_SIZE=0 RX_BUFFER_SIZE=32 TX_BUFFER_SIZE=32)
 if(MINGW)
  target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
 endif()
endforeach()
set_source_files_properties("${driver}/Src/serial.c" PROPERTIES COMPILE_OPTIONS -Wno-unused-parameter)

set_tests_properties(driver_stm32f1_serial.capacity PROPERTIES WILL_FAIL TRUE LABELS known_bug)
