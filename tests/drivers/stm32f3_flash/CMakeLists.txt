cmake_minimum_required(VERSION 3.20)
project(grblhal_stm32f3_flash_tests LANGUAGES C)
include(../../components.cmake)
foreach(behavior erased_read geometry128 geometry256 full_page unchanged unlock_failure erase_failure program_first_failure program_second_failure retry)
 set(case_source ${behavior}.c)
 if(behavior MATCHES "^geometry(128|256)$")
  set(case_source geometry.c)
  set(capacity ${CMAKE_MATCH_1})
 endif()
 component_case(driver_stm32f3_flash.${behavior} ${case_source} f3_host.c "${HARNESS_ROOT}/drivers/stm32f3/Src/flash.c")
 if(behavior MATCHES "^geometry")
  target_compile_definitions(driver_stm32f3_flash.${behavior} PRIVATE F3_FLASH_KB=${capacity})
 endif()
 target_compile_options(driver_stm32f3_flash.${behavior} PRIVATE -m32)
 target_link_options(driver_stm32f3_flash.${behavior} PRIVATE -m32)
 if(NOT WIN32)
  target_compile_definitions(driver_stm32f3_flash.${behavior} PRIVATE _GNU_SOURCE)
 endif()
endforeach()

set_tests_properties(driver_stm32f3_flash.program_first_failure PROPERTIES WILL_FAIL TRUE LABELS known_bug)
