cmake_minimum_required(VERSION 3.20)
project(grblhal_ti_shared_eeprom_tests LANGUAGES C)
include(../../components.cmake)
set(source "${HARNESS_ROOT}/drivers/ti_shared/base/eeprom.c")
set(host "${HARNESS_ROOT}/tests/drivers/ti_eeprom")
foreach(crc 1 2)
  set(behaviors checksum_layout checksum_read)
  if(crc EQUAL 2)
    list(APPEND behaviors initialization byte_lanes block_alignment program_failure allocation_write_failure allocation_read_failure zero_length)
  endif()
  foreach(behavior IN LISTS behaviors)
    set(target driver_ti_shared_eeprom.crc${crc}_${behavior})
    component_case(${target} "${host}/${behavior}.c" "${host}/ti_host.c" "${COMPONENT_CORE_ROOT}/crc.c" "${source}")
    target_include_directories(${target} PRIVATE "${host}")
    target_include_directories(${target} SYSTEM PRIVATE "${HARNESS_ROOT}/deps/tivaware")
    target_compile_definitions(${target} PRIVATE OVERRIDE_MY_MACHINE=1 BOARD_CNC_BOOSTERPACK=1 PART_TM4C1294NCPDT CNC_BOOSTERPACK_A4998=1 NVS_CRC_BYTES=${crc})
  endforeach()
endforeach()
# Inject allocation failure only at the original EEPROM module's heap boundary.
set_source_files_properties("${source}" PROPERTIES COMPILE_DEFINITIONS malloc=ti_malloc)


foreach(behavior program_failure allocation_write_failure allocation_read_failure)
  set_tests_properties(driver_ti_shared_eeprom.crc2_${behavior} PROPERTIES WILL_FAIL TRUE LABELS known_bug)
endforeach()
foreach(crc 1 2)
  foreach(behavior checksum_layout checksum_read)
    set_tests_properties(driver_ti_shared_eeprom.crc${crc}_${behavior} PROPERTIES WILL_FAIL TRUE LABELS known_bug)
  endforeach()
endforeach()
