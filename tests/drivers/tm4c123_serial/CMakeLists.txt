cmake_minimum_required(VERSION 3.20)
project(grblhal_tm4c123_serial_tests LANGUAGES C)
include(../../components.cmake)
set(core_sources)
foreach(file stream.c stream_passthru.c vfs.c strutils.c nuts_bolts.c)
  list(APPEND core_sources "${COMPONENT_CORE_ROOT}/${file}")
  set_source_files_properties("${COMPONENT_CORE_ROOT}/${file}" PROPERTIES COMPILE_OPTIONS -Wno-error)
endforeach()
foreach(behavior initialization rx_fifo rx_wrap rx_cancel rts_flow full_realtime direct_tx queued_tx full_tx_abort blocking_tx suspend handler_replace)
  set(target driver_tm4c123_serial.${behavior})
  component_case(${target} ${behavior}.c uart_host.c core_boundaries.c "${HARNESS_ROOT}/drivers/tm4c123/serial.c" ${core_sources})
  target_include_directories(${target} PRIVATE "${HARNESS_ROOT}/drivers/tm4c123")
  target_include_directories(${target} SYSTEM PRIVATE "${HARNESS_ROOT}/deps/tivaware")
  target_compile_definitions(${target} PRIVATE OVERRIDE_MY_MACHINE=1 BOARD_CNC_BOOSTERPACK=1 PART_TM4C123GH6PM RX_BUFFER_SIZE=1024 TX_BUFFER_SIZE=32 MPG_BUFFER_SIZE=0)
  if(MINGW)
    target_compile_options(${target} PRIVATE -include "${HARNESS_ROOT}/tests/include/windows_string_compat.h")
  endif()
endforeach()

set_tests_properties(driver_tm4c123_serial.full_realtime PROPERTIES WILL_FAIL TRUE LABELS known_bug)
