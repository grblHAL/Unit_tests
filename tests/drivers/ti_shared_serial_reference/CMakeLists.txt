cmake_minimum_required(VERSION 3.20)
project(grblhal_ti_shared_serial_reference LANGUAGES C)
get_filename_component(reference_core "${CMAKE_CURRENT_SOURCE_DIR}/../../../deps/core-20250518" ABSOLUTE)
set(COMPONENT_CORE_ROOT "${reference_core}" CACHE PATH "Explicit TI shared serial legacy CORE reference")
include(../../components.cmake)
set(host "${HARNESS_ROOT}/tests/drivers/tm4c123_serial")
foreach(behavior handler_restore initialization rx_fifo rx_wrap rx_cancel full_realtime tx_queue backpressure disable claim)
  if(behavior MATCHES "^(initialization|rx_fifo|rx_wrap|rx_cancel|full_realtime)$")
    set(test_source "${host}/${behavior}.c")
  else()
    set(test_source "${behavior}.c")
  endif()
  set(target driver_ti_shared_serial_reference.${behavior})
  component_case(${target} "${test_source}" "${host}/uart_host.c" reference_boundaries.c "${HARNESS_ROOT}/drivers/ti_shared/base/serial.c")
  target_include_directories(${target} PRIVATE "${host}" "${HARNESS_ROOT}/drivers/ti_shared/base")
  target_include_directories(${target} SYSTEM PRIVATE "${HARNESS_ROOT}/deps/tivaware" "${COMPONENT_CORE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/include")
  target_compile_definitions(${target} PRIVATE TI_SHARED_REFERENCE=1 OVERRIDE_MY_MACHINE=1 BOARD_CNC_BOOSTERPACK=1 PART_TM4C1294NCPDT CNC_BOOSTERPACK_A4998=1)
endforeach()

foreach(signedness signed unsigned)
  set(target driver_ti_shared_serial_reference.stream_suspend_${signedness})
  component_case(${target} legacy_suspend.c "${HARNESS_ROOT}/tests/drivers/legacy_stream_boundaries.c" "${host}/uart_host.c" "${HARNESS_ROOT}/drivers/ti_shared/base/serial.c" "${COMPONENT_CORE_ROOT}/stream.c")
  target_include_directories(${target} PRIVATE "${host}" "${HARNESS_ROOT}/drivers/ti_shared/base")
  target_include_directories(${target} SYSTEM PRIVATE "${HARNESS_ROOT}/deps/tivaware" "${COMPONENT_CORE_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/include")
  target_compile_definitions(${target} PRIVATE TI_SHARED_REFERENCE=1 LEGACY_CORE_STREAM=1 OVERRIDE_MY_MACHINE=1 BOARD_CNC_BOOSTERPACK=1 PART_TM4C1294NCPDT CNC_BOOSTERPACK_A4998=1)
  target_compile_options(${target} PRIVATE -f${signedness}-char)
endforeach()
set_source_files_properties("${COMPONENT_CORE_ROOT}/stream.c" PROPERTIES COMPILE_OPTIONS -Wno-error)

set_tests_properties(driver_ti_shared_serial_reference.stream_suspend_signed PROPERTIES WILL_FAIL TRUE LABELS known_bug)
