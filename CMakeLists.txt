function(add_host_core target)
  add_library(${target} STATIC EXCLUDE_FROM_ALL tests/support/engine_host.c tests/support/task_host.c)
  target_link_libraries(${target} PRIVATE grbl)
  target_include_directories(${target} PUBLIC core tests)
  target_compile_features(${target} PRIVATE c_std_11)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  target_compile_definitions(${target} PUBLIC ${HOST_API_DEFINITIONS} ${ARGN})
  if(ENABLE_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
  if(ENABLE_COVERAGE)
    target_compile_options(${target} PRIVATE --coverage -O0 -g)
    target_link_options(${target} PUBLIC --coverage)
  endif()
endfunction()

    elseif(module STREQUAL "homing_command_uvw")
      target_link_libraries(test_${name} PRIVATE host_core_uvw)
    elseif(module STREQUAL "motion_mode_tracking")
      target_link_libraries(test_${name} PRIVATE host_core_motion_mode)
    elseif(module STREQUAL "debug_serial")
      target_link_libraries(test_${name} PRIVATE host_core_debug_serial)
    elseif(module STREQUAL "debug_external")
      target_link_libraries(test_${name} PRIVATE host_core_debug_external)
    elseif(module STREQUAL "nvs_debug")
      target_link_libraries(test_${name} PRIVATE host_core_nvs_debug)
    elseif(module STREQUAL "nvs_debug_eight")
      target_link_libraries(test_${name} PRIVATE host_core_nvs_debug_eight)

      target_link_libraries(test_${name} PRIVATE host_core_tools)

  elseif(module STREQUAL "rgb")
    # Header-only driver utilities: exercise the original inline definitions.
    add_executable(test_${name} ${source})

    target_compile_options(test_${name} PRIVATE /W4)
  else()
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror)
    if(module STREQUAL "machine_limits")
      target_compile_options(test_${name} PRIVATE -ffunction-sections -fdata-sections)
      target_link_options(test_${name} PRIVATE -Wl,--gc-sections)
    endif()
    # Keep our tests strict without making upstream warnings test failures.
    set_source_files_properties(core/${module}.c PROPERTIES COMPILE_OPTIONS -Wno-error)
    target_link_libraries(test_${name} PRIVATE m)
    if(ENABLE_SANITIZERS)
      target_compile_options(test_${name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(test_${name} PRIVATE -fsanitize=address,undefined)
    endif()
    if(ENABLE_COVERAGE)
      if(NOT CMAKE_C_COMPILER_ID MATCHES "^(GNU|Clang)$")
        message(FATAL_ERROR "Coverage requires GCC or GNU-compatible Clang")
      endif()
      target_compile_options(test_${name} PRIVATE --coverage -O0 -g)
      target_link_options(test_${name} PRIVATE --coverage)
    endif()
  endif()
  if(name STREQUAL "report.alarm_catalog_allocation_failure_retains_every_registered_entry"
    OR name STREQUAL "report.error_catalog_allocation_failure_retains_every_registered_entry"
    OR name STREQUAL "stream_json.start_allocation_failure_preserves_output_and_allows_retry"
    OR name STREQUAL "report.group_metadata_fallback_keeps_visible_settings"
    OR name STREQUAL "report.setting_group_fallback_lists_available_groups")

  endif()
  if(name MATCHES "^stream\\.auxiliary_connection_")

  if(name STREQUAL "gcode.motion_comment_releases_all_message_allocations_after_planner_reset")

  endif()
  if(name MATCHES "^planner\\.allocation_" OR name MATCHES "^boot\\.planner_allocation_")
    target_link_options(test_${name} PRIVATE -Wl,--wrap=malloc)
  endif()
  if(name STREQUAL "encoders.registration_allocation_failure_preserves_count_and_allows_retry"
    OR name STREQUAL "report.pins_report_survives_sort_allocation_failure"
    OR name STREQUAL "report.uart_report_survives_sort_allocation_failure"
    OR name STREQUAL "ngc_params.virtual_registration_allocation_failure_preserves_existing_list")
    target_link_options(test_${name} PRIVATE -Wl,--wrap=malloc)
  endif()
  if(name STREQUAL "ngc_params.string_growth_allocation_failure_preserves_neighbors_and_allows_retry"
    OR name STREQUAL "ngc_params.string_insert_allocation_failure_preserves_existing_values"
    OR name STREQUAL "ngc_params.string_name_allocation_failure_reports_zero_and_allows_retry")
    target_link_options(test_${name} PRIVATE -Wl,--wrap=realloc)
  endif()
  if(name STREQUAL "nvs_buffer.buffer_allocation_failure_can_retry_without_changing_storage_handlers"
    OR name STREQUAL "nvs_buffer.driver_allocation_failure_can_retry_without_reserving_area"
    OR name STREQUAL "nvs_buffer.initialization_without_buffer_reports_startup_warning")
    target_link_options(test_${name} PRIVATE -Wl,--wrap=malloc)
  endif()
  if(name STREQUAL "vfs.current_directory_reallocation_failure_falls_back_to_root")
    target_link_options(test_${name} PRIVATE -Wl,--wrap=free)
  endif()
  if(name STREQUAL "vfs.current_directory_reallocation_failure_falls_back_to_root"
    OR name STREQUAL "vfs.current_directory_allocation_failure_falls_back_to_root"
    OR name STREQUAL "vfs.drive_iterator_allocation_failure_can_retry"
    OR name STREQUAL "vfs.mount_allocation_failure_reports_failure_without_notification"
    OR name STREQUAL "vfs.mount_retry_after_allocation_failure_registers_one_backend")
    target_link_options(test_${name} PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
  endif()
  if(name MATCHES "^motion_control\\.thread_plan_")
    target_link_options(test_${name} PRIVATE -Wl,--wrap=plan_buffer_line)
  endif()
  add_test(NAME ${name} COMMAND test_${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 15)
endfunction()
